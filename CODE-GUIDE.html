CREATE TABLE IF NOT EXISTS users (
 user_id INT AUTO_INCREMENT PRIMARY KEY,
 username VARCHAR(30) NOT NULL UNIQUE,
 email VARCHAR(190) NOT NULL UNIQUE,
 password VARCHAR(255) NOT NULL,
 fullname VARCHAR(100) NOT NULL,
 bio TEXT NOT NULL,
 avatar VARCHAR(64) NULL,
 created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;
CREATE TABLE IF NOT EXISTS categories (
 category_id INT AUTO_INCREMENT PRIMARY KEY,
 name VARCHAR(100) NOT NULL UNIQUE
) ENGINE=InnoDB;
CREATE TABLE IF NOT EXISTS trips (
 trip_id INT AUTO_INCREMENT PRIMARY KEY,
 user_id INT NOT NULL,
 category_id INT NOT NULL,
 title VARCHAR(150) NOT NULL,
 description TEXT NOT NULL,
 start_date DATE NOT NULL,
 end_date DATE NOT NULL,
 visibility ENUM('public','private') NOT NULL DEFAULT 'private',
 cover VARCHAR(64) NOT NULL,
 created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
 updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
 FOREIGN KEY(user_id) REFERENCES users(user_id) ON DELETE CASCADE,
 FOREIGN KEY(category_id) REFERENCES categories(category_id),
 INDEX dates_visibility(visibility,start_date,end_date),
 INDEX owner(user_id)
) ENGINE=InnoDB;
CREATE TABLE IF NOT EXISTS trip_places (
 place_id INT AUTO_INCREMENT PRIMARY KEY,
 trip_id INT NOT NULL,
 name VARCHAR(150) NOT NULL,
 province VARCHAR(100) NOT NULL,
 visit_date DATE NOT NULL,
 story TEXT NOT NULL,
 sort_order INT NOT NULL DEFAULT 1,
 FOREIGN KEY(trip_id) REFERENCES trips(trip_id) ON DELETE CASCADE
) ENGINE=InnoDB;
CREATE TABLE IF NOT EXISTS place_photos (
 photo_id INT AUTO_INCREMENT PRIMARY KEY,
 place_id INT NOT NULL,
 filename VARCHAR(64) NOT NULL UNIQUE,
 FOREIGN KEY(place_id) REFERENCES trip_places(place_id) ON DELETE CASCADE
) ENGINE=InnoDB;
CREATE TABLE IF NOT EXISTS bookmarks (
 user_id INT NOT NULL,
 trip_id INT NOT NULL,
 created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
 PRIMARY KEY(user_id,trip_id),
 FOREIGN KEY(user_id) REFERENCES users(user_id) ON DELETE CASCADE,
 FOREIGN KEY(trip_id) REFERENCES trips(trip_id) ON DELETE CASCADE
) ENGINE=InnoDB;

function get_trip($trip_id, $owner_only = false)
{
    global $conn;
    $stmt = $conn->prepare(
        "SELECT t.*, u.fullname, u.username, u.avatar, c.name AS category_name FROM trips t JOIN users u ON t.user_id=u.user_id JOIN categories c ON t.category_id=c.category_id WHERE t.trip_id=?",
    );
    $stmt->bind_param("i", $trip_id);
    $stmt->execute();
    $trip = $stmt->get_result()->fetch_assoc();
    $stmt->close();
    $owner = $trip && (int) $trip["user_id"] === user_id();
    if (
        !$trip ||
        ($owner_only && !$owner) ||
        ($trip["visibility"] === "private" && !$owner)
    ) {
        not_found();
    }
    return $trip;
}
function get_place($place_id, $owner_only = false)
{
    global $conn;
    $stmt = $conn->prepare("SELECT * FROM trip_places WHERE place_id=?");
    $stmt->bind_param("i", $place_id);
    $stmt->execute();
    $place = $stmt->get_result()->fetch_assoc();
    $stmt->close();
    if (!$place) {
        not_found();
    }
    $trip = get_trip((int) $place["trip_id"], $owner_only);
    return [$place, $trip];
}

function text_value($name, $max, $required = false)
{
    $value = $_POST[$name] ?? "";
    if (!is_string($value)) {
        throw new RuntimeException("รูปแบบข้อมูลไม่ถูกต้อง");
    }
    $value = trim($value);
    if (($required && $value === "") || mb_strlen($value) > $max) {
        throw new RuntimeException(
            "กรุณากรอกข้อมูลให้ครบและไม่เกินความยาวที่กำหนด",
        );
    }
    return $value;
}
function valid_date($value)
{
    if (!is_string($value)) {
        return false;
    }
    $d = DateTimeImmutable::createFromFormat("!Y-m-d", $value);
    return $d &&
        $d->format("Y-m-d") === $value &&
        $value >= "1900-01-01" &&
        $value <= "2200-12-31";
}
function trip_input()
{
    global $conn;
    $title = text_value("title", 150, true);
    $description = text_value("description", 5000);
    $start = text_value("start_date", 10, true);
    $end = text_value("end_date", 10, true);
    $visibility = text_value("visibility", 10, true);
    $category = (int) ($_POST["category_id"] ?? 0);
    if (!valid_date($start) || !valid_date($end) || $start > $end) {
        throw new RuntimeException("ช่วงวันที่เดินทางไม่ถูกต้อง");
    }
    if (!in_array($visibility, ["public", "private"], true)) {
        throw new RuntimeException("กรุณาเลือกความเป็นส่วนตัว");
    }
    $stmt = $conn->prepare(
        "SELECT category_id FROM categories WHERE category_id=?",
    );
    $stmt->bind_param("i", $category);
    $stmt->execute();
    if (!$stmt->get_result()->fetch_assoc()) {
        throw new RuntimeException("ไม่พบประเภทการท่องเที่ยว");
    }
    $stmt->close();
    return [$title, $description, $start, $end, $visibility, $category];
}
function place_input($trip)
{
    $name = text_value("name", 150, true);
    $province = text_value("province", 100, true);
    $date = text_value("visit_date", 10, true);
    $story = text_value("story", 10000);
    $order = (int) ($_POST["sort_order"] ?? 1);
    if (
        !valid_date($date) ||
        $date < $trip["start_date"] ||
        $date > $trip["end_date"]
    ) {
        throw new RuntimeException("วันที่ไปเยือนต้องอยู่ในช่วงการเดินทาง");
    }
    if ($order < 1 || $order > 1000) {
        throw new RuntimeException("ลำดับต้องอยู่ระหว่าง 1–1000");
    }
    return [$name, $province, $date, $story, $order];
}

function upload_photo($file, &$created, $required = false)
{
    $error = (int) ($file["error"] ?? UPLOAD_ERR_NO_FILE);
    if ($error === UPLOAD_ERR_NO_FILE && !$required) {
        return null;
    }
    if ($error !== UPLOAD_ERR_OK) {
        throw new RuntimeException("กรุณาเลือกรูป หรือรูปอัปโหลดไม่สำเร็จ");
    }
    if (
        (int) $file["size"] > 5 * 1024 * 1024 ||
        !is_uploaded_file($file["tmp_name"])
    ) {
        throw new RuntimeException("ไฟล์รูปแต่ละไฟล์ต้องไม่เกิน 5 MB");
    }
    $mime = (new finfo(FILEINFO_MIME_TYPE))->file($file["tmp_name"]);
    $types = [
        "image/jpeg" => "jpg",
        "image/png" => "png",
        "image/webp" => "webp",
    ];
    $dimensions = @getimagesize($file["tmp_name"]);
    if (
        !isset($types[$mime]) ||
        !$dimensions ||
        $dimensions[0] * $dimensions[1] > 25000000
    ) {
        throw new RuntimeException(
            "รองรับ JPG PNG WebP และขนาดภาพไม่เกิน 25 ล้านพิกเซล",
        );
    }
    $name = bin2hex(random_bytes(24)) . "." . $types[$mime];
    if (!move_uploaded_file($file["tmp_name"], UPLOAD_DIR . "/" . $name)) {
        throw new RuntimeException("เขียนไฟล์รูปภาพไม่ได้");
    }
    $created[] = $name;
    return $name;
}
function add_photos($place_id, &$created)
{
    global $conn;
    $files = $_FILES["photos"] ?? [];
    $errors = $files["error"] ?? [];
    $count = count(photos_for($place_id));
    foreach ($errors as $i => $error) {
        if ((int) $error === UPLOAD_ERR_NO_FILE) {
            continue;
        }
        if (++$count > 10) {
            throw new RuntimeException("แต่ละสถานที่มีรูปได้ไม่เกิน 10 รูป");
        }
        $file = [
            "error" => $error,
            "size" => $files["size"][$i] ?? 0,
            "tmp_name" => $files["tmp_name"][$i] ?? "",
        ];
        $filename = upload_photo($file, $created, true);
        $stmt = $conn->prepare(
            "INSERT INTO place_photos(place_id,filename) VALUES(?,?)",
        );
        $stmt->bind_param("is", $place_id, $filename);
        $stmt->execute();
        $stmt->close();
    }
}
function bookmark_trip($trip_id)
{
    global $conn;
    $uid = user_id();
    $stmt = $conn->prepare(
        "SELECT user_id FROM trips WHERE trip_id=? AND visibility='public'",
    );
    $stmt->bind_param("i", $trip_id);
    $stmt->execute();
    $trip = $stmt->get_result()->fetch_assoc();
    $stmt->close();
    if (!$trip || (int) $trip["user_id"] === $uid) {
        return false;
    }
    $stmt = $conn->prepare(
        "INSERT IGNORE INTO bookmarks(user_id,trip_id) VALUES(?,?)",
    );
    $stmt->bind_param("ii", $uid, $trip_id);
    $stmt->execute();
    $stmt->close();
    return true;
}

โครงการนี้ใช้รูปแบบเดียวกับไฟล์ตัวอย่าง: PHP ส่วนประมวลผลอยู่ด้านบน ตามด้วย HTML, ใช้ MySQLi `$conn->prepare()` / `bind_param()` / `get_result()`, Session และ CSS ปกติ ไม่ใช้ React, Node.js, PDO หรือ framework ในตัวเว็บไซต์

## ติดตั้งบน XAMPP

2. เปิด Apache และ MySQL ใน XAMPP
3. เปิด `http://localhost/phpmyadmin` → Import → เลือก `database/schema.sql` → Go

## รูปภาพและการตั้งค่า PHP

ปกติเมื่อวางโปรเจกต์ใน `xampp/htdocs/paibaekpea` ระบบเก็บรูปที่ `xampp/paibaekpea-private-uploads` (นอก htdocs) หากใช้โครงสร้างโฟลเดอร์อื่น ให้ตั้ง UPLOAD_DIR ใน environment หรือแก้ `$privatePath` ใน includes/init.php เป็นพาธนอก document root

รับ JPG/PNG/WebP ตรวจ MIME ด้วย fileinfo และขนาดภาพด้วย getimagesize จำกัด 5 MiB / 25 ล้านพิกเซลต่อไฟล์; ไม่เกิน 10 รูปต่อสถานที่ และ 20 สถานที่ต่อทริป ไม่มีการ re-encode หรือลบ EXIF ในรุ่นนี้

ใน php.ini แนะนำสำหรับการสาธิต:

```ini
upload_max_filesize = 5M
post_max_size = 64M
max_file_uploads = 20
```

เปลี่ยนแล้ว restart Apache หากเกิน post_max_size PHP อาจไม่ส่ง $_POST/$_FILES และระบบจะแจ้งคำขอหมดอายุ ให้ลดจำนวนไฟล์ที่อัปโหลดในครั้งเดียว

สำรองทั้งฐานข้อมูล MySQL และโฟลเดอร์รูปภาพภายนอกโปรเจกต์ อย่า backup เฉพาะ htdocs

## โครงสร้างและความปลอดภัย

- ใช้ prepared statements กับข้อมูลผู้ใช้; escape HTML ก่อนแสดง
- password_hash/password_verify; session_regenerate_id หลังล็อกอิน
- CSRF token ทุก mutation รวมลบ/ออกจากระบบ
- ตรวจเจ้าของที่ server สำหรับแก้ไขและลบทั้งทริปและสถานที่
- รูปภาพไม่ใช้ชื่อเดิมจากผู้ใช้งาน ใช้ random_bytes ตั้งชื่อใหม่
- transaction สำหรับทริป/สถานที่; rollback พร้อมลบไฟล์ใหม่เมื่อคำขอล้มเหลว; ลบไฟล์เก่าหลัง commit
- CSP ไม่อนุญาต JavaScript จึงใช้หน้ายืนยันและ form POST แบบ PHP ล้วน
- จำกัด login ผิด 10 ครั้ง/15 นาทีต่อ Session เท่านั้น ยังไม่ใช่ rate limiter ต่อ IP หรือระบบป้องกัน brute force สำหรับ production

        # ฐานข้อมูล

users (user_id PK, username/email UNIQUE) → trips (trip_id PK, user_id/category_id FK) → trip_places (place_id PK, trip_id FK) → place_photos (photo_id PK, place_id FK, filename UNIQUE)

categories 1:N trips; users M:N trips ผ่าน bookmarks ซึ่งใช้ composite primary key (user_id,trip_id) ป้องกันซ้ำ มี ON DELETE CASCADE สำหรับทริป สถานที่ รูป และ Bookmark

cover อยู่ใน trips และ avatar อยู่ใน users เป็นชื่อไฟล์สุ่มของระบบ เก็บไฟล์จริงนอก web root ทุกภาพเข้าผ่าน media.php
